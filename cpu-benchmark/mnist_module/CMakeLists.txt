cmake_minimum_required(VERSION 3.19)
project(cpumnistmodule LANGUAGES CXX)
if(NOT NEST_PREFIX)
  message(FATAL_ERROR "Set NEST_PREFIX to the source-built NEST installation")
endif()
execute_process(COMMAND ${NEST_PREFIX}/bin/nest-config --includes OUTPUT_VARIABLE nest_includes OUTPUT_STRIP_TRAILING_WHITESPACE)
separate_arguments(nest_includes NATIVE_COMMAND "${nest_includes}")
add_library(cpumnistmodule MODULE module.cpp)
set(CPU_MNIST_INPUT_COUNT 784 CACHE STRING "Maximum number of image input neurons")
target_compile_features(cpumnistmodule PRIVATE cxx_std_20)
target_compile_definitions(cpumnistmodule PRIVATE LTX_MODULE CPU_MNIST_INPUT_COUNT=${CPU_MNIST_INPUT_COUNT})
target_compile_options(cpumnistmodule PRIVATE ${nest_includes} -O3 -march=native -fopenmp)
target_link_options(cpumnistmodule PRIVATE -fopenmp)
set_target_properties(cpumnistmodule PROPERTIES PREFIX "")
