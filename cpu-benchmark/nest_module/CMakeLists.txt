cmake_minimum_required(VERSION 3.19)
project(cpubrunelmodule LANGUAGES CXX)
if(NOT NEST_PREFIX)
  message(FATAL_ERROR "Set NEST_PREFIX to the source-built NEST installation")
endif()
execute_process(COMMAND ${NEST_PREFIX}/bin/nest-config --includes OUTPUT_VARIABLE nest_includes OUTPUT_STRIP_TRAILING_WHITESPACE)
separate_arguments(nest_includes NATIVE_COMMAND "${nest_includes}")
add_library(cpubrunelmodule MODULE module.cpp)
target_compile_features(cpubrunelmodule PRIVATE cxx_std_20)
target_compile_definitions(cpubrunelmodule PRIVATE LTX_MODULE)
target_compile_options(cpubrunelmodule PRIVATE ${nest_includes} -O3 -march=native -fopenmp)
target_link_options(cpubrunelmodule PRIVATE -fopenmp)
set_target_properties(cpubrunelmodule PROPERTIES PREFIX "")
